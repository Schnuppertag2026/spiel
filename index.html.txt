const worldSize = 2500;
const segments = 500;
const terrainGeo = new THREE.PlaneGeometry(worldSize, worldSize, segments, segments);

function organicHeight(x,y){
    const s1 = 0.002, s2 = 0.006, s3 = 0.0012;
    const bigHills  = Math.sin(x*s1)*50 + Math.cos(y*s1*1.2)*40;
    const detail    = Math.sin((x+y)*s2)*12 + Math.cos((x-y)*s2*1.4)*8;
    const longWaves = Math.sin(x*s3)*30 * Math.cos(y*s3*0.7);
    const distToCenter = Math.sqrt(x*x + y*y);
    const basin = -Math.max(0, 60 - distToCenter*0.04);
    return bigHills + detail + longWaves + basin;
}

for(let i=0;i<terrainGeo.attributes.position.count;i++){
    const x = terrainGeo.attributes.position.getX(i);
    const y = terrainGeo.attributes.position.getY(i);
    terrainGeo.attributes.position.setZ(i, organicHeight(x,y));
}
terrainGeo.computeVertexNormals();

const terrainMat = new THREE.MeshStandardMaterial({
    color: 0x228B22,
    roughness: 0.85,
    metalness: 0.02
});
const terrain = new THREE.Mesh(terrainGeo, terrainMat);
terrain.rotation.x = -Math.PI/2;
scene.add(terrain);

const GRAVITY = 0.5;
function getTerrainHeight(x,z){
    const px = ((x+worldSize/2)/worldSize)*segments;
    const pz = ((z+worldSize/2)/worldSize)*segments;
    const ix = Math.floor(px);
    const iz = Math.floor(pz);
    const idx = iz*(segments+1)+ix;
    if(idx<0 || idx>=terrainGeo.attributes.position.count) return 0;
    return terrainGeo.attributes.position.getZ(idx);
}
function isLand(x,z){ return getTerrainHeight(x,z) >= waterLevel + 1; }

// === BERGE ===
function createBorderMountains(){
    const mountainMat = new THREE.MeshStandardMaterial({color:0x666666, roughness:0.95});
    const ringRadius = worldSize/2 + 200;
    const ringWidth  = 400;
    const segmentsRing = 120;
    const ringGeo = new THREE.PlaneGeometry(ringRadius*2, ringRadius*2, segmentsRing, segmentsRing);

    for(let i=0;i<ringGeo.attributes.position.count;i++){
        const vx = ringGeo.attributes.position.getX(i);
        const vy = ringGeo.attributes.position.getY(i);
        const dist = Math.sqrt(vx*vx + vy*vy);
        const inner = worldSize/2;
        const outer = worldSize/2 + ringWidth;
        let h = 0;
        if(dist > inner && dist < outer){
            const t = (dist - inner)/(outer - inner);
            const base = 200 + Math.random()*150;
            const sharp = Math.pow(Math.sin(t*Math.PI), 1.5);
            h = base * sharp + (Math.sin(vx*0.01)*Math.cos(vy*0.01))*40;
        }
        ringGeo.attributes.position.setZ(i, h);
    }
    ringGeo.computeVertexNormals();
    const ringMesh = new THREE.Mesh(ringGeo, mountainMat);
    ringMesh.rotation.x = -Math.PI/2;
    ringMesh.position.y = waterLevel - 5;
    scene.add(ringMesh);
}
createBorderMountains();

const player = {mesh:null, speed:1.8, vy:0, onGround:false, jumpPower:10, isMoving:false, animTime:0};
let playerParts = [];

function createPlayerModel(){
    const group = new THREE.Group();

    const bodyGeo = new THREE.BoxGeometry(0.8,1.8,0.5);
    const bodyMat = new THREE.MeshStandardMaterial({color:0x3366ff});
    const body = new THREE.Mesh(bodyGeo, bodyMat);
    body.position.y = 1.8;
    group.add(body);
    playerParts.push({mesh:body, type:'body'});

    const headGeo = new THREE.BoxGeometry(0.45,0.45,0.45);
    const headMat = new THREE.MeshStandardMaterial({color:0xffdbac});
    const head = new THREE.Mesh(headGeo, headMat);
    head.position.y = 2.6;
    group.add(head);
    playerParts.push({mesh:head, type:'head'});

    const armGeo = new THREE.BoxGeometry(0.2,0.9,0.25);
    const armMat = new THREE.MeshStandardMaterial({color:0x3366ff});
    const leftArm = new THREE.Mesh(armGeo, armMat);
    leftArm.position.set(-0.55,1.65,0);
    const rightArm = new THREE.Mesh(armGeo, armMat);
    rightArm.position.set(0.55,1.65,0);
    group.add(leftArm);
    group.add(rightArm);
    playerParts.push({mesh:leftArm, type:'arm'});
    playerParts.push({mesh:rightArm, type:'arm'});

    const legGeo = new THREE.BoxGeometry(0.25,1.2,0.3);
    const legMat = new THREE.MeshStandardMaterial({color:0x222222});
    const leftLeg = new THREE.Mesh(legGeo, legMat);
    leftLeg.position.set(-0.2,0.6,0);
    const rightLeg = new THREE.Mesh(legGeo, legMat);
    rightLeg.position.set(0.2,0.6,0);
    group.add(leftLeg);
    group.add(rightLeg);
    playerParts.push({mesh:leftLeg, type:'leg'});
    playerParts.push({mesh:rightLeg, type:'leg'});

    return group;
}

player.mesh = createPlayerModel();
player.mesh.position.set(0,40,0);
scene.add(player.mesh);

function updatePlayerAnimation(deltaTime){
    player.animTime += deltaTime * 0.005;
    playerParts.forEach((part, index) => {
        if(part.type === 'leg'){
            const legIndex = playerParts.indexOf(part);
            const swing = Math.sin(player.animTime * 10 + legIndex) * 0.4;
            part.mesh.rotation.z = swing * (player.isMoving ? 1 : 0.2);
            part.mesh.rotation.x = Math.abs(swing) * 0.25 * (player.isMoving ? 1 : 0.2);
        }
        else if(part.type === 'arm'){
            const armIndex = playerParts.indexOf(part);
            const swing = Math.sin(player.animTime * 10 + (armIndex % 2)) * 0.35;
            part.mesh.rotation.z = -swing * (player.isMoving ? 1 : 0.2);
        }
    });
}

function createRandomAnimalMesh(){
    const animal = new THREE.Group();
    const bodyGeo = new THREE.BoxGeometry(1.2,0.6,2.2);
    const bodyMat = new THREE.MeshStandardMaterial({color:0x8b5a2b});
    const body = new THREE.Mesh(bodyGeo, bodyMat);
    body.position.y = 0.6;
    animal.add(body);

    const headGeo = new THREE.BoxGeometry(0.6,0.55,0.6);
    const head = new THREE.Mesh(headGeo, bodyMat);
    head.position.set(0,0.8,1.2);
    animal.add(head);

    const legGeo = new THREE.BoxGeometry(0.2,0.55,0.2);
    const legMat = new THREE.MeshStandardMaterial({color:0x4b2e15});
    const legPositions = [
        [-0.4,0.275,0.6], [0.4,0.275,0.6],
        [-0.4,0.275,-0.7], [0.4,0.275,-0.7]
    ];
    legPositions.forEach(p=>{
        const leg = new THREE.Mesh(legGeo, legMat);
        leg.position.set(p[0],p[1],p[2]);
        animal.add(leg);
    });
    return animal;
}

        changeTime: Date.now() + 2000 + Math.random()*4000
    });
}

        const height = 5 + Math.random()*3;

        const trunkGeo = new THREE.CylinderGeometry(0.25,0.35,trunkHeight,6);
        const trunkMat = new THREE.MeshStandardMaterial({color:0x5b3a1a});

        const crownMat = new THREE.MeshStandardMaterial({color:0x0b6623});

        crown.position.y = trunkHeight/2 + height*0.325;

        const tree = new THREE.Group();
        tree.add(trunk);
        tree.add(crown);

        let x,z,y;
        do{

        tree.position.set(x, y, z);
        scene.add(tree);
    }

    for(let i=0;i<300;i++){

const keys={};
document.addEventListener('keydown', e=>keys[e.code]=true);
document.addEventListener('keyup', e=>keys[e.code]=false);

let pointerDown=false,pointerX=0,pointerY=0;
let cameraAngleX=0,cameraAngleY=0;
function startPointer(e){
    pointerDown=true; 
    pointerX=e.clientX||e.touches[0].clientX; 
    pointerY=e.clientY||e.touches[0].clientY;
}
function movePointer(e){
    if(!pointerDown) return;
    const x=e.clientX||e.touches[0].clientX;
    const y=e.clientY||e.touches[0].clientY;
    cameraAngleX-=(x-pointerX)*0.005;
    cameraAngleY-=(y-pointerY)*0.005;
    cameraAngleY=Math.max(-Math.PI/3,Math.min(Math.PI/3,cameraAngleY));
    pointerX=x; pointerY=y;
}
function endPointer(){pointerDown=false;}
document.addEventListener('mousedown',startPointer);
document.addEventListener('mousemove',movePointer);
document.addEventListener('mouseup',endPointer);
document.addEventListener('touchstart',startPointer);
document.addEventListener('touchmove',movePointer);
document.addEventListener('touchend',endPointer);

function updateCamera(){
    const px = player.mesh.position.x, py = player.mesh.position.y, pz = player.mesh.position.z;
    const radius = 25;
    const camX = px + radius * Math.sin(cameraAngleX) * Math.cos(cameraAngleY);
    const camY = py + radius * Math.sin(cameraAngleY) + 10;
    const camZ = pz + radius * Math.cos(cameraAngleX) * Math.cos(cameraAngleY);
    camera.position.set(camX, camY, camZ);
    camera.lookAt(player.mesh.position);

    const forward = new THREE.Vector3();
    camera.getWorldDirection(forward); forward.y = 0; forward.normalize();
    const right = new THREE.Vector3();
    right.crossVectors(forward, new THREE.Vector3(0,1,0)).normalize();

    let moveDir = new THREE.Vector3();
    if(keys["KeyW"]) moveDir.add(forward);
    if(keys["KeyS"]) moveDir.add(forward.clone().negate());
    if(keys["KeyA"]) moveDir.add(right.clone().negate());
    if(keys["KeyD"]) moveDir.add(right);

    player.isMoving = moveDir.lengthSq() > 0;
    
    if(moveDir.lengthSq()>0){
        moveDir.normalize().multiplyScalar(player.speed);
        player.mesh.position.add(moveDir);
    }
}

function updateAnimals(deltaTime){
    const now = Date.now();
    for(let i=0;i<animals.length;i++){
        const a = animals[i];
        const state = animalStates[i];
        if(now > state.changeTime){
            const angleChange = (Math.random()-0.5) * Math.PI/2;
            const currentAngle = Math.atan2(state.dir.x, state.dir.z);
            const newAngle = currentAngle + angleChange;
            state.dir.set(Math.sin(newAngle),0,Math.cos(newAngle));
            state.speed = 0.25 + Math.random()*0.4;
            state.changeTime = now + 1500 + Math.random()*4000;
        }
        const move = state.dir.clone().multiplyScalar(state.speed * deltaTime*0.06);
        a.position.add(move);
        const border = worldSize*0.45;
        if(Math.abs(a.position.x) > border || Math.abs(a.position.z) > border || !isLand(a.position.x,a.position.z)){
            state.dir.multiplyScalar(-1);
        }
        const terrainY = getTerrainHeight(a.position.x,a.position.z)+0.7;
        a.position.y = terrainY + Math.sin(now*0.003 + i)*0.1;
        const angle = Math.atan2(state.dir.x, state.dir.z);
        a.rotation.y = angle;
    }
}

// === HAUPTSCHLEIFE ===
let lastTime = performance.now();
function loop(){
    const currentTime = performance.now();
    const delta = currentTime - lastTime;
    lastTime = currentTime;

    if(keys["Space"] && player.onGround){
        player.vy=player.jumpPower;
        player.onGround=false;
    }
    const groundY=getTerrainHeight(player.mesh.position.x,player.mesh.position.z)+1.3;
    if(player.mesh.position.y>groundY){
        player.vy-=GRAVITY;
    }else{
        player.vy=0;
        player.mesh.position.y=groundY;
        player.onGround=true;
    }
    player.mesh.position.y += player.vy;

    updatePlayerAnimation(delta);
    updateAnimals(delta);

    for(let i=animals.length-1;i>=0;i--){
        const a = animals[i];
        if(player.mesh.position.distanceTo(a.position)<2){
            scene.remove(a);
            animals.splice(i,1);
            animalStates.splice(i,1);
            caughtUI.textContent = ANIMAL_COUNT-animals.length;
        }
    }

    updateCamera();
    renderer.render(scene,camera);
    requestAnimationFrame(loop);
}
loop();
